import os

User = get_user_model()


class Command(BaseCommand):
    help = (
        "Trusted server-side bootstrap command to provision or promote the initial "
        "HMS Administrator account with RoleProfile.Role.ADMINISTRATOR."
    )

    def add_arguments(self, parser):
        parser.add_argument(
            "--username",
            required=True,
            help="Username for the administrator account.",
        )
        parser.add_argument(
            "--email",
            required=True,
            help="University email address for the administrator account.",
        )
        parser.add_argument(
            "--password",
            default=None,
            help=(
                "Password for the administrator account. If omitted, reads from "
                "the HMS_BOOTSTRAP_ADMIN_PASSWORD environment variable."
            ),
        )
        parser.add_argument(
            "--first-name",
            default="System",
            help="First name for the administrator account (default: System).",
        )
        parser.add_argument(
            "--last-name",
            default="Administrator",
            help="Last name for the administrator account (default: Administrator).",
        )

    def handle(self, *args, **options):
        username = options["username"].strip()
        email = options["email"].strip().lower()
        password = options["password"] or os.environ.get("HMS_BOOTSTRAP_ADMIN_PASSWORD")
        first_name = options["first_name"].strip()
        last_name = options["last_name"].strip()

        if not username:
            raise CommandError("A non-empty --username is required.")
        if not email or "@" not in email:
            raise CommandError("A valid --email address is required.")
        if not password:
            raise CommandError(
                "Password must be supplied via --password or the HMS_BOOTSTRAP_ADMIN_PASSWORD environment variable."
            )

        candidate_user = User(
            username=username,
            email=email,
            first_name=first_name,
            last_name=last_name,
        )
        try:
            validate_password(password, user=candidate_user)
        except ValidationError as exc:
            raise CommandError("; ".join(exc.messages)) from exc

        with transaction.atomic():
            user = User.objects.filter(username__iexact=username).first()
            is_new_user = user is None

            if is_new_user:
                user = User.objects.create_user(
                    username=username,
                    email=email,
                    password=password,
                    first_name=first_name,
                    last_name=last_name,
                    is_active=True,
                    is_staff=True,
                    is_superuser=True,
                )
            else:
                user.email = email
                user.first_name = first_name
                user.last_name = last_name
                user.is_active = True
                user.is_staff = True
                user.is_superuser = True
                user.set_password(password)
                user.save()

            profile = RoleProfile.objects.filter(user=user).first()
            is_new_profile = profile is None
            if is_new_profile:
                profile = RoleProfile(user=user)

            profile.role = RoleProfile.Role.ADMINISTRATOR
            profile.student = None
            profile.save()

            create_audit_log(
                user,
                "CREATE" if is_new_profile else "UPDATE",
                "RoleProfile",
                profile.pk,
            )

        action_label = "Created" if is_new_user else "Updated"
        self.stdout.write(
            self.style.SUCCESS(
                f"{action_label} HMS Administrator '{user.username}' ({user.email}) with active ADMINISTRATOR role."
            )
        )

import { type NextRequest, NextResponse } from "next/server";

function getBackendOrigin(): string {
  return (process.env.BACKEND_ORIGIN || "http://127.0.0.1:8000").replace(
    /\/+$/,
    ""
  );
}

function normalizeSetCookieForProxy(rawCookie: string): string {
  // Remove any explicit Domain attribute so the browser stores the cookie
  // for the frontend origin (e.g. localhost:3000) and sends it on subsequent
  // same-origin /api/v1/* requests.
  return rawCookie.replace(/;\s*Domain=[^;]*/gi, "");
}

async function proxyToDjango(
  request: NextRequest,
  context: { params: Promise<{ path: string[] }> }
): Promise<NextResponse> {
  const { path } = await context.params;
  const backendOrigin = getBackendOrigin();
  const cleanSegments = (path || []).filter(Boolean);
  const search = request.nextUrl.search || "";

  // Always terminate the Django API path with a trailing slash before any query string
  const targetUrl = `${backendOrigin}/api/v1/${cleanSegments.join("/")}/${search}`;

  const upstreamHeaders = new Headers();
  upstreamHeaders.set(
    "Accept",
    request.headers.get("accept") || "application/json"
  );

  const contentType = request.headers.get("content-type");
  if (contentType) {
    upstreamHeaders.set("Content-Type", contentType);
  }

  const cookieHeader = request.headers.get("cookie");
  if (cookieHeader) {
    upstreamHeaders.set("Cookie", cookieHeader);
  }

  const csrfHeader = request.headers.get("x-csrftoken");
  if (csrfHeader) {
    upstreamHeaders.set("X-CSRFToken", csrfHeader);
  }

  const originHeader = request.headers.get("origin");
  if (originHeader) {
    upstreamHeaders.set("Origin", originHeader);
  }

  const refererHeader = request.headers.get("referer");
  if (refererHeader) {
    upstreamHeaders.set("Referer", refererHeader);
  }

  const method = request.method.toUpperCase();
  let body: ArrayBuffer | undefined;
  if (method !== "GET" && method !== "HEAD") {
    const buffer = await request.arrayBuffer();
    if (buffer.byteLength > 0) {
      body = buffer;
    }
  }

  let upstreamResponse: Response;
  try {
    upstreamResponse = await fetch(targetUrl, {
      method,
      headers: upstreamHeaders,
      body,
      redirect: "manual",
      cache: "no-store",
    });
  } catch {
    return NextResponse.json(
      {
        error:
          "Unable to reach the HMS backend API (/api/v1). Ensure the Django backend server is running.",
        details: {},
        status_code: 503,
      },
      { status: 503 }
    );
  }

  const responseHeaders = new Headers();
  const upstreamContentType = upstreamResponse.headers.get("content-type");
  if (upstreamContentType) {
    responseHeaders.set("Content-Type", upstreamContentType);
  }

  const upstreamCsrfToken = upstreamResponse.headers.get("x-csrftoken");
  if (upstreamCsrfToken) {
    responseHeaders.set("X-CSRFToken", upstreamCsrfToken);
  }

  responseHeaders.set("Cache-Control", "no-store, no-cache, must-revalidate");

  const setCookies =
    typeof upstreamResponse.headers.getSetCookie === "function"
      ? upstreamResponse.headers.getSetCookie()
      : [];

  if (setCookies.length > 0) {
    for (const cookie of setCookies) {
      responseHeaders.append("Set-Cookie", normalizeSetCookieForProxy(cookie));
    }
  } else {
    const fallbackSetCookie = upstreamResponse.headers.get("set-cookie");
    if (fallbackSetCookie) {
      responseHeaders.append(
        "Set-Cookie",
        normalizeSetCookieForProxy(fallbackSetCookie)
      );
    }
  }

  if (upstreamResponse.status === 204) {
    return new NextResponse(null, {
      status: 204,
      headers: responseHeaders,
    });
  }

  const responseBody = await upstreamResponse.arrayBuffer();
  return new NextResponse(responseBody, {
    status: upstreamResponse.status,
    headers: responseHeaders,
  });
}

export async function GET(
  request: NextRequest,
  context: { params: Promise<{ path: string[] }> }
): Promise<NextResponse> {
  return proxyToDjango(request, context);
}

export async function POST(
